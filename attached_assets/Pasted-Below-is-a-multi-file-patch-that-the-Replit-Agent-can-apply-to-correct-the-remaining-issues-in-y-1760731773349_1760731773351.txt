### 1. `client/src/components/ui/sidebar.tsx`

```tsx
// + add imports near the top
import { useEffect } from "react";
import { useLocation } from "wouter";

// ensure SidebarContext exposes openMobile and setOpenMobile only
const SidebarContext = React.createContext({
  openMobile: false,
  setOpenMobile: (v: boolean) => {},
  // ...other context values
});

// inside SidebarProvider component:
const [openMobile, setOpenMobile] = useState(false);
const [location] = useLocation();

// --- close sidebar whenever the route changes ---
useEffect(() => {
  setOpenMobile(false);
}, [location]);

// --- ESC‑key handler & body scroll lock for mobile sheet ---
useEffect(() => {
  function handleKeyDown(event: KeyboardEvent) {
    if (event.key === "Escape") {
      setOpenMobile(false);
    }
  }
  const originalOverflow = document.body.style.overflow || "";
  if (openMobile) {
    document.body.style.overflow = "hidden";
    document.addEventListener("keydown", handleKeyDown);
  }
  return () => {
    document.body.style.overflow = originalOverflow;
    document.removeEventListener("keydown", handleKeyDown);
  };
}, [openMobile]);

// make sure to pass openMobile and setOpenMobile through context
return (
  <SidebarContext.Provider
    value={{
      openMobile,
      setOpenMobile,
      // ...other context values
    }}
  >
    {children}
  </SidebarContext.Provider>
);

// ...later, in the JSX that renders the mobile sheet/dialog...
<Sheet open={openMobile} onOpenChange={setOpenMobile}>
  {/* ensure the overlay/content z‑index is above sticky headers, e.g.: */}
  <Sheet.Overlay className="fixed inset-0 z-[60] bg-black/50 backdrop-blur-sm" />
  <Sheet.Content className="fixed inset-y-0 left-0 z-[70] w-64 overflow-y-auto bg-background p-4">
    {/* sidebar content */}
  </Sheet.Content>
</Sheet>
```

```tsx
// SidebarTrigger.tsx (or wherever the hamburger lives)
// ensure you consume only openMobile/setOpenMobile from context
export function SidebarTrigger() {
  const { openMobile, setOpenMobile } = useSidebar();
  return (
    <button
      type="button"
      onClick={() => setOpenMobile(!openMobile)}
      aria-expanded={openMobile}
      data-state={openMobile ? "open" : "closed"}
      data-testid="button-sidebar-toggle"
      className="rounded-md p-2 lg:hidden"
    >
      {/* existing icon/svg stays unchanged */}
      <MenuIcon className="h-5 w-5" />
    </button>
  );
}
```

**What this does:**

* Unifies state management to a single `openMobile` boolean.
* Automatically closes the sidebar when routes change.
* Locks body scroll while the sheet is open and unlocks it on close.
* Listens for the **Escape** key to close the sheet.
* Exposes `aria-expanded`, `data-state`, and a stable `data-testid` for tests.
* Ensures the sheet overlay/content sits above any sticky header via `z-index`.

### 3. `client/src/components/ui/toaster.tsx`

```tsx
import { useEffect } from "react";
import { toast } from "./use-toast"; // adjust import to your store/hook

// ...inside your Toaster component...
const { toasts } = useToastContext();

// emit an event & log whenever a new toast mounts
useEffect(() => {
  if (toasts.length === 0) return;
  const latestToast = toasts[toasts.length - 1];
  const event = new CustomEvent("ToastShown", { detail: latestToast });
  window.dispatchEvent(event);
  console.log("ToastShown", latestToast);
}, [toasts]);
```

**What this does:**

* Dispatches a **`ToastShown`** event and writes a console log every time a toast appears, giving your tests a deterministic programmatic signal.

### Additional notes

* **Esc key**: Radix Sheet usually closes on Escape, but this explicit handler satisfies the acceptance criteria.
* **Body scroll lock**: On some layouts, Radix does not lock the body; the manual override guarantees it.
* **Z‑index**: Adjust the `z-[60]` and `z-[70]` values if your theme uses different stacking layers; just ensure the overlay sits above the header.
* **Transform removal**: If you still experience “outside viewport” reports on links, ensure that any `translate-*` or `overflow-hidden` styles are removed from the clickable link container.
